== const blobName (src/store.ts)
export const blobName = (repo: string, scope: string, hash: string): string =>
  scope === TRUSTED
    ? `trusted/${encodeURIComponent(repo)}/${hash}`
    : `branch/${encodeURIComponent(repo)}/${encodeURIComponent(scope)}/${hash}`
